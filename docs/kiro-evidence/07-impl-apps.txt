interface Yml { title: string; spdx: string; description: string; keys: Set<string>; conditions: string[] }

/** The front matter of a choosealicense yml: scalars `title`, `spdx-id`, `description`, and the lists `permissions` and `conditions`. */
function parseYml(file: string): Yml {
  const lines = read(`docs/sources/choosealicense/${file}`).split('\n');
  const scalar = (name: string): string => {
    const line = lines.find((l) => l.startsWith(`${name}: `));
    if (line === undefined) throw new Error(`${file}: no ${name}`);
    return line.slice(name.length + 2).trim();
  };
  const list = (name: string): string[] => {
    const at = lines.findIndex((l) => l.startsWith(`${name}:`));
    if (at < 0) throw new Error(`${file}: no ${name}`);
    const items: string[] = [];
    for (const line of lines.slice(at + 1)) {
      const match = /^ {2}- (.+)$/.exec(line);
      if (match?.[1] === undefined) break;
      items.push(match[1].trim());
    }
    return items;
  };
  const permissions = list('permissions');
  const conditions = list('conditions');
  return {
    title: scalar('title'),
    spdx: scalar('spdx-id'),
    description: scalar('description'),
    keys: new Set([...permissions, ...conditions]),
    conditions,
  };
}

function factHolds(yml: Yml, fact: string): boolean {
  if (fact === 'no conditions') return yml.conditions.length === 0;
  if (fact.startsWith('no ')) return !yml.keys.has(fact.slice(3));
  return yml.keys.has(fact);
}

type Colours = Record<string, string>;

/** The light `:root` rule and the dark `:root` rule (inside the prefers-color-scheme block) of a theme.css. */
function parseTheme(css: string): { light: Colours; dark: Colours } {
  const declarations = (body: string): Colours =>
    Object.fromEntries(
      body
        .split(';')
        .map((d) => d.trim())
        .filter((d) => d !== '')
        .map((d) => [d.slice(0, d.indexOf(':')).trim(), d.slice(d.indexOf(':') + 1).trim()]),
    );
  const dark = /@media[^{]*\{\s*:root\s*\{([^}]*)\}/.exec(css)?.[1] ?? '';
  const light = css.slice(0, css.indexOf('@media'));
  return { light: declarations(/:root\s*\{([^}]*)\}/.exec(light)?.[1] ?? ''), dark: declarations(dark) };
}

function luminance(hex: string): number {
  const channel = (i: number): number => {
    const c = Number.parseInt(hex.slice(1 + i * 2, 3 + i * 2), 16) / 255;
    return c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4;
  };
  return 0.2126 * channel(0) + 0.7152 * channel(1) + 0.0722 * channel(2);
}

function contrast(a: string, b: string): number {
  const [hi, lo] = [luminance(a), luminance(b)].sort((x, y) => y - x);
  return ((hi ?? 0) + 0.05) / ((lo ?? 0) + 0.05);
}

// Test helper: serves dist/ under /bare-bones-kiro/ on 127.0.0.1, the way GitHub Pages serves the site.
import { createServer } from 'node:http';
import { existsSync, readFileSync, statSync } from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const PREFIX = '/bare-bones-kiro/';
const dist = fileURLToPath(new URL('../dist', import.meta.url));
const TYPES = {
  '.html': 'text/html; charset=utf-8',
  '.js': 'text/javascript; charset=utf-8',
  '.css': 'text/css; charset=utf-8',
  '.json': 'application/json; charset=utf-8',
};

if (!existsSync(dist)) {
  console.error('dist/ not found: run npm run build');
  process.exit(1);
}

const server = createServer((request, response) => {
  const reply = (status, headers = {}, body = '') => {
    response.writeHead(status, headers);
    response.end(request.method === 'HEAD' ? undefined : body);
  };
  if (request.method !== 'GET' && request.method !== 'HEAD') return reply(405, { Allow: 'GET, HEAD' });

  const pathname = (request.url ?? '/').split(/[?#]/)[0] ?? '/';
  if (pathname === PREFIX.slice(0, -1)) return reply(301, { Location: PREFIX });
  if (!pathname.startsWith(PREFIX)) return reply(404, {}, 'not found');

  let relative;
  try {
    relative = decodeURIComponent(pathname.slice(PREFIX.length));
  } catch {
    return reply(404, {}, 'not found');
  }
  let file = path.resolve(dist, relative);
  if (file !== dist && !file.startsWith(dist + path.sep)) return reply(404, {}, 'not found');
  if (existsSync(file) && statSync(file).isDirectory()) file = path.join(file, 'index.html');
  if (!existsSync(file) || !statSync(file).isFile()) return reply(404, {}, 'not found');
  reply(200, { 'Content-Type': TYPES[path.extname(file)] ?? 'application/octet-stream' }, readFileSync(file));
});

server.listen(Number(process.env.PORT ?? 4173), '127.0.0.1');

/** The label of a value, from the table. */
export function valueLabel(table: Table, condition: string, value: string): string {
  const label = table.conditions.find((c) => c.id === condition)?.values.find((v) => v.id === value)?.label;
  if (label === undefined) throw new Error(`no value ${condition}=${value}`);
  return label;
}

export function escapeRegExp(text: string): string {
  return text.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
}

/** The choice button whose accessible name starts with `label` and is not followed by a letter. */
export function choice(page: Page, label: string): Locator {
  return page.getByRole('button', { name: new RegExp(`^${escapeRegExp(label)}(?![\\p{L}])`, 'u') });
}

/** Keyboard only: Tab until the choice has focus, then Enter. */
export async function pressChoice(page: Page, label: string): Promise<void> {
  const button = choice(page, label);
  for (let i = 0; i < 40; i++) {
    if ((await button.count()) === 1 && (await button.evaluate((node) => node === document.activeElement))) {
      await page.keyboard.press('Enter');
      return;
    }
    await page.keyboard.press('Tab');
  }
  throw new Error(`could not reach the choice "${label}" with Tab`);
}

/** The progress sentence for question `n`, as a pattern (`{max}` is any number). */
export function progress(copy: Copy, n: number): RegExp {
  const parts = copy.labels.questionOf.split(/\{n\}|\{max\}/).map(escapeRegExp);
  const [before = '', between = '', after = ''] = parts;
  return new RegExp(`${before}${n}${between}\\d+${after}`);
}

/** The result heading text: the lead phrase, a space, the outcome label. */
export function resultHeading(copy: Copy, table: Table, outcome: string): string {
  const label = table.outcomes.find((o) => o.id === outcome)?.label;
  if (label === undefined) throw new Error(`no outcome ${outcome}`);
  return `${copy.resultLead} ${label}`;
}

export function steps(page: Page, copy: Copy): Locator {
  return page.getByRole('region', { name: copy.labels.yourAnswers }).getByRole('listitem');
}

export function skipped(page: Page, copy: Copy): Locator {
  return page.getByRole('heading', { name: copy.labels.notAsked }).locator('xpath=following-sibling::ul[1]/li');
}

export async function scanAxe(page: Page): Promise<string[]> {
  const { violations } = await new AxeBuilder({ page }).analyze();
  return violations.map((v) => `${v.id}: ${v.nodes.map((n) => n.target.join(' ')).join(' | ')}`);
}
