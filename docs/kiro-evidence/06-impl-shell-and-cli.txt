export type Child = Node | string;

/** Creates an element. Text goes in as text nodes and attributes through setAttribute, never as markup. */
export function el<K extends keyof HTMLElementTagNameMap>(
  tag: K,
  attrs: Record<string, string> = {},
  ...children: Child[]
): HTMLElementTagNameMap[K] {
  const node = document.createElement(tag);
  for (const [name, value] of Object.entries(attrs)) node.setAttribute(name, value);
  for (const child of children) node.appendChild(typeof child === 'string' ? document.createTextNode(child) : child);
  return node;
}

export function text(value: string): Text {
  return document.createTextNode(value);
}

/** Replaces each `{name}` that has a value. Unknown names are left as they are. */
export function fill(template: string, values: Record<string, string | number>): string {
  return template.replace(/\{([a-z]+)\}/g, (whole, name: string) =>
    Object.hasOwn(values, name) ? String(values[name]) : whole,
  );
}

/** The URL if it is http(s), otherwise undefined. */
export function safeHref(url: string): string | undefined {
  return /^https?:\/\//i.test(url) ? url : undefined;
}

/** A link when the URL is safe, otherwise plain text. */
export function linkOrText(label: string, url: string): Node {
  const href = safeHref(url);
  return href === undefined ? text(label) : el('a', { href, rel: 'noopener noreferrer' }, label);
}

import type { Table } from '../engine/index.ts';

export interface Step { condition: string; value: string }

/** `#condition=value&…` in step order; empty for no steps. Ids are kebab-case, so nothing is encoded. */
export function writeHash(steps: readonly Step[]): string {
  if (steps.length === 0) return '';
  return '#' + steps.map((s) => `${s.condition}=${s.value}`).join('&');
}

/** Keeps the entries that name a known condition and one of its values, once each. Never throws. */
export function readHash(table: Table, hash: string): { steps: Step[]; ignored: number } {
  const body = hash.startsWith('#') ? hash.slice(1) : hash;
  if (body === '') return { steps: [], ignored: 0 };

  const values = new Map<string, Set<string>>();
  for (const condition of table.conditions) values.set(condition.id, new Set(condition.values.map((v) => v.id)));

  const steps: Step[] = [];
  const taken = new Set<string>();
  let ignored = 0;
  for (const part of body.split('&')) {
    const pieces = part.split('=');
    const [condition, value] = pieces;
    const allowed = condition === undefined ? undefined : values.get(condition);
    if (
      pieces.length !== 2 ||
      condition === undefined ||
      value === undefined ||
      condition === '' ||
      value === '' ||
      allowed === undefined ||
      !allowed.has(value) ||
      taken.has(condition)
    ) {
      ignored += 1;
      continue;
    }
    taken.add(condition);
    steps.push({ condition, value });
  }
  return { steps, ignored };
}

import type { AppSummary, Copy, Landing } from '../shell/copy-type.ts';

const KEY_PATTERN = /^[a-z0-9]+(-[a-z0-9]+)*$/;

/** JSON text that is safe inside a `<script type="application/json">` element. */
export function escapeJson(value: unknown): string {
  return (JSON.stringify(value) ?? 'null')
    .replaceAll('<', '\\u003c')
    .replaceAll('>', '\\u003e')
    .replaceAll('&', '\\u0026')
    .replaceAll('\u2028', '\\u2028')
    .replaceAll('\u2029', '\\u2029');
}

export function escapeHtml(value: string): string {
  return value
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;')
    .replaceAll("'", '&#39;');
}

/** The static block for an app page: shown without JavaScript, removed by the script at start. */
export function bootApp(copy: Copy): string {
  return (
    '<div id="boot" class="page">' +
    `<h1>${escapeHtml(copy.title)}</h1>` +
    `<p>${escapeHtml(copy.tagline)}</p>` +
    `<p>${escapeHtml(copy.intro)}</p>` +
    `<noscript><p>${escapeHtml(copy.labels.noscript)}</p></noscript>` +
    '</div>'
  );
}

/** The static block for the landing page: the title, the tagline and a link per app. */
export function bootLanding(landing: Landing, apps: readonly AppSummary[]): string {
  const items = apps
    .map((a) => `<li><a href="./${escapeHtml(a.id)}/">${escapeHtml(a.title)}</a><p>${escapeHtml(a.tagline)}</p></li>`)
    .join('');
  return (
    '<div id="boot" class="page">' +
    `<h1>${escapeHtml(landing.title)}</h1>` +
    `<p>${escapeHtml(landing.tagline)}</p>` +
    `<ul class="apps">${items}</ul>` +
    '</div>'
  );
}

/** Puts `boot` and one JSON data block per entry of `blocks` into the template. */
export function renderPage(template: string, blocks: Record<string, unknown>, boot: string): string {
  const data = Object.entries(blocks)
    .map(([key, value]) => {
      if (!KEY_PATTERN.test(key)) throw new Error(`bad data block key: ${key}`);
      return `<script type="application/json" id="data-${key}">${escapeJson(value)}</script>`;
    })
    .join('\n');
  return template.replace('<!--bones:boot-->', () => boot).replace('<!--bones:data-->', () => data);
}

export async function readRepoJson(repoRoot: string, p: string): Promise<unknown> {
  // Lexical check first, so a probe cannot learn whether a file outside the repository exists.
  const absolute = path.resolve(repoRoot, p);
  if (outside(repoRoot, absolute)) throw new Error('path is outside the repository');

    // V8's message quotes part of the input; keep only the position.
    const position = error instanceof Error ? /at position \d+(?: \(line \d+ column \d+\))?/.exec(error.message)?.[0] : undefined;
    throw new Error(position === undefined ? 'invalid JSON' : `invalid JSON ${position}`);
  }
}

function errorText(error: unknown): string {
  return error instanceof Error ? error.message : 'unknown error';
}

export function createServer(repoRoot: string): McpServer {
  const server = new McpServer({ name: 'bare-bones', version: '0.1.0' });

  server.registerTool(
    'check_table',
    {
      description:
        'Check a decision table for structural problems, gaps and conflicts. Give exactly one of `path` (a .json file inside the repository) or `table` (the table itself).',
      inputSchema: { path: z.string().optional(), table: z.unknown().optional() },
      annotations: { readOnlyHint: true },
    },
    async ({ path: file, table }) => {
      if ((file === undefined) === (table === undefined)) {
        return reply({ error: 'give exactly one of path and table' }, true);
      }
      let input: unknown = table;
      if (file !== undefined) {
        try {
          input = await readRepoJson(repoRoot, file);
        } catch (error) {
          return reply({ error: errorText(error) }, true);
        }
      }
      const result = validate(input);
      if (!result.ok) return reply({ valid: false, problems: result.problems });
      return reply({ valid: true, report: check(result.table) });
    },
  );

  server.registerTool(
    'evaluate',
    {
      description:
        'Give answers (condition id to value id) to a table that passes the check, and get the decision: decided with the outcome and why, or open with the next question.',
      inputSchema: { path: z.string(), answers: z.record(z.string(), z.string()) },
      annotations: { readOnlyHint: true },
    },
    async ({ path: file, answers }) => {
      let input: unknown;
      try {
        input = await readRepoJson(repoRoot, file);
      } catch (error) {
        return reply({ error: errorText(error) }, true);
      }
      const result = validate(input);
      if (!result.ok) return reply({ valid: false, passed: false, problems: result.problems }, true);
      const report = check(result.table);
      if (!report.passed) {
        return reply({ valid: true, passed: false, gaps: report.gaps.count, conflicts: report.conflicts.length }, true);
      }
      return reply(decide(result.table, answers));
    },
  );

  return server;
}
