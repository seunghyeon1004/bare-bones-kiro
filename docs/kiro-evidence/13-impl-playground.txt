export interface JsonError {
  reason: 'end' | 'character' | 'string' | 'trailing';
  offset: number; // 0-based UTF-16 index of the first bad position
  line: number; // 1-based; a line break is "\n"
  column: number; // 1-based, in UTF-16 code units from the start of the line
  character?: string; // the offending character, for 'character'
}

const WHITESPACE = new Set([' ', '\t', '\n', '\r']);
const ESCAPES = new Set(['"', '\\', '/', 'b', 'f', 'n', 'r', 't']);
const DIGIT = /[0-9]/;
const HEX = /[0-9a-fA-F]/;

type Frame = 'array' | 'object';
/** What the scanner expects next. */
type Expect = 'value' | 'value-or-close' | 'key' | 'key-or-close' | 'colon' | 'comma-or-close';

function position(text: string, offset: number): { line: number; column: number } {
  let line = 1;
  let lineStart = 0;
  for (let i = 0; i < offset && i < text.length; i++) {
    if (text[i] === '\n') {
      line++;
      lineStart = i + 1;
    }
  }
  return { line, column: offset - lineStart + 1 };
}

/**
 * The first syntax error of `text` under the JSON grammar of RFC 8259, or undefined when the text is valid.
 * Iterative (an explicit stack), so deeply nested input cannot overflow the call stack.
 */
export function locateJsonError(text: string): JsonError | undefined {
  const n = text.length;
  let i = 0;
  const fail = (reason: JsonError['reason'], offset: number): JsonError => {
    const at = Math.min(offset, n);
    const error: JsonError = { reason, offset: at, ...position(text, at) };
    if (reason === 'character' && at < n) error.character = text.charAt(at);
    return error;
  };
  const skipSpace = (): void => {
    while (i < n && WHITESPACE.has(text.charAt(i))) i++;
  };

  /** Reads a string starting at the opening quote; returns an error or undefined. */
  const readString = (): JsonError | undefined => {
    i++;
    for (;;) {
      if (i >= n) return fail('end', n);
      const c = text.charAt(i);
      if (c === '"') {
        i++;
        return undefined;
      }
      if (c.charCodeAt(0) < 0x20) return fail('string', i);
      if (c === '\\') {
        const e = text.charAt(i + 1);
        if (i + 1 >= n) return fail('end', n);
        if (ESCAPES.has(e)) {
          i += 2;
        } else if (e === 'u') {
          for (let k = 2; k <= 5; k++) {
            if (i + k >= n) return fail('end', n);
            if (!HEX.test(text.charAt(i + k))) return fail('string', i + k);
          }
          i += 6;
        } else {
          return fail('string', i + 1);
        }
      } else {
        i++;
      }
    }
  };

  const readLiteral = (word: string): JsonError | undefined => {
    for (let k = 0; k < word.length; k++) {
      if (i + k >= n) return fail('end', n);
      if (text.charAt(i + k) !== word.charAt(k)) return fail('character', i + k);
    }
    i += word.length;
    return undefined;
  };

  const readNumber = (): JsonError | undefined => {
    const digits = (): JsonError | undefined => {
      if (i >= n) return fail('end', n);
      if (!DIGIT.test(text.charAt(i))) return fail('character', i);
      while (i < n && DIGIT.test(text.charAt(i))) i++;
      return undefined;
    };
    if (text.charAt(i) === '-') i++;
    if (i >= n) return fail('end', n);
    if (text.charAt(i) === '0') {
      i++;
    } else {
      const error = digits();
      if (error !== undefined) return error;
    }
    if (text.charAt(i) === '.') {
      i++;
      const error = digits();
      if (error !== undefined) return error;
    }
    if (text.charAt(i) === 'e' || text.charAt(i) === 'E') {
      i++;
      if (text.charAt(i) === '+' || text.charAt(i) === '-') i++;
      const error = digits();
      if (error !== undefined) return error;
    }
    return undefined;
  };

  const stack: Frame[] = [];
  let expect: Expect = 'value';
  let done = false;

  while (!done) {
    skipSpace();
    if (i >= n) return fail('end', n);
    const c = text.charAt(i);

    if (expect === 'key' || expect === 'key-or-close') {
      if (c === '}' && expect === 'key-or-close') {
        i++;
        stack.pop();
        expect = 'comma-or-close';
      } else if (c === '"') {
        const error = readString();
        if (error !== undefined) return error;
        expect = 'colon';
      } else {
        return fail('character', i);
      }
    } else if (expect === 'colon') {
      if (c !== ':') return fail('character', i);
      i++;
      expect = 'value';
    } else if (expect === 'comma-or-close') {
      const top = stack[stack.length - 1];
      if (top === undefined) {
        done = true;
      } else if (c === ',') {
        i++;
        expect = top === 'array' ? 'value' : 'key';
      } else if ((c === ']' && top === 'array') || (c === '}' && top === 'object')) {
        i++;
        stack.pop();
      } else {
        return fail('character', i);
      }
    } else {
      // 'value' or 'value-or-close'
      if (c === ']' && expect === 'value-or-close') {
        i++;
        stack.pop();
        expect = 'comma-or-close';
      } else if (c === '{') {
        i++;
        stack.push('object');
        expect = 'key-or-close';
      } else if (c === '[') {
        i++;
        stack.push('array');
        expect = 'value-or-close';
      } else {
        let error: JsonError | undefined;
        if (c === '"') error = readString();
        else if (c === 't') error = readLiteral('true');
        else if (c === 'f') error = readLiteral('false');
        else if (c === 'n') error = readLiteral('null');
        else if (c === '-' || DIGIT.test(c)) error = readNumber();
        else return fail('character', i);
        if (error !== undefined) return error;
        expect = 'comma-or-close';
      }
    }
    if (expect === 'comma-or-close' && stack.length === 0) done = true;
  }

  skipSpace();
  return i < n ? fail('trailing', i) : undefined;
}

import { check, validate } from '../engine/index.ts';
import type { CheckReport, Problem, Table } from '../engine/index.ts';
import { locateJsonError } from './json-error.ts';
import type { JsonError } from './json-error.ts';

export type Analysis =
  | { kind: 'syntax'; error: JsonError }
  | { kind: 'structure'; problems: Problem[] }
  | { kind: 'checked'; table: Table; report: CheckReport };

/** Reads table text: not JSON, JSON with structural problems, or checked. The order `npm run build` uses for a table. */
export function analyse(text: string): Analysis {
  let parsed: unknown;
  try {
    parsed = JSON.parse(text);
  } catch {
    // JSON.parse decides that the text is invalid; the scanner only places the error.
    // If the two ever disagree, still report an error rather than none.
    return { kind: 'syntax', error: locateJsonError(text) ?? { reason: 'character', offset: 0, line: 1, column: 1 } };
  }
  const result = validate(parsed);
  if (!result.ok) return { kind: 'structure', problems: result.problems };
  return { kind: 'checked', table: result.table, report: check(result.table) };
}

export function passes(analysis: Analysis): boolean {
  return analysis.kind === 'checked' && analysis.report.passed;

        gaps: plural(p.gap, report.gaps.count),
        conflicts: plural(p.conflict, report.conflicts.length),
      }),
    ),

  };

export interface RuleRow { index: number; id: string | undefined }

export function formatTable(table: unknown): string {
  return JSON.stringify(table, null, 2);
}

function parseRules(text: string): { parsed: Record<string, unknown>; rules: unknown[] } | undefined {
  let parsed: unknown;
  try {
    parsed = JSON.parse(text);
  } catch {
    return undefined;
  }
  if (typeof parsed !== 'object' || parsed === null || Array.isArray(parsed)) return undefined;
  const object = parsed as Record<string, unknown>;
  const rules = object['rules'];
  return Array.isArray(rules) ? { parsed: object, rules } : undefined;
}

/** One row per element of `rules`; undefined unless the text is a JSON object with a `rules` array. */
export function listRules(text: string): RuleRow[] | undefined {
  const found = parseRules(text);
  if (found === undefined) return undefined;
  return found.rules.map((rule, index) => {
    const id = typeof rule === 'object' && rule !== null && !Array.isArray(rule) ? (rule as Record<string, unknown>)['id'] : undefined;
    return { index, id: typeof id === 'string' ? id : undefined };
  });
}

/** The text without the rule at `index` (by position), re-serialised with two-space indentation. */
export function removeRule(text: string, index: number): string {
  const found = parseRules(text);
  if (found === undefined || !Number.isInteger(index) || index < 0 || index >= found.rules.length) return text;
  found.rules.splice(index, 1);
  return formatTable(found.parsed);

export interface ParsedTheme { light: Record<string, string>; dark: Record<string, string> }

/**
 * The custom properties of a theme.css as `{ light, dark }`: the declarations of the top-level `:root` and of the `:root`
 * inside the dark `@media` block. Call it on a theme that passed `validateTheme`.
 */
export function parseTheme(css: string): ParsedTheme {
  const text = css.replace(/\/\*[\s\S]*?\*\//g, '');
  const theme: ParsedTheme = { light: {}, dark: {} };
  const stack: string[] = [];
  let buffer = '';
  for (const ch of text) {
    if (ch === '{') {
      stack.push(buffer.trim().replace(/\s+/g, ' '));
      buffer = '';
    } else if (ch === '}') {
      const prelude = stack.pop();
      if (prelude === ':root') {
        const target = stack.length === 0 ? theme.light : theme.dark;
        for (const raw of buffer.split(';')) {
          const colon = raw.indexOf(':');
          if (colon > 0) target[raw.slice(0, colon).trim()] = raw.slice(colon + 1).trim();
        }
      }
      buffer = '';
    } else {
      buffer += ch;
    }
  }
  return theme;
}

    structureHeading: [],
    gapsHeading: [],
    gapsNone: [],
    conflictsHeading: [],
    conflictsNone: [],
    conflictLine: ['{a}', '{b}', '{outcomeA}', '{outcomeB}', '{example}'],
    more: ['{count}'],
    warningsHeading: [],
  },
  plural: { rule: PLURAL, combination: PLURAL, gap: PLURAL, conflict: PLURAL, problem: PLURAL },
  app: { heading: [], noBuildHeading: [], noBuildText: [], noBuildReason: ['{reason}'] },
  footer: 'footer',
} satisfies Record<keyof PlaygroundCopy, Shape>;

function isObj(x: unknown): x is Obj {
  return typeof x === 'object' && x !== null && !Array.isArray(x);
}

function walk(shape: Shape, value: unknown, path: string, problems: string[]): void {
  if (shape === 'footer') {
    checkFooter(value, problems);
  } else if (Array.isArray(shape)) {
    if (typeof value !== 'string') {
      problems.push(`${path}: must be a string`);
    } else if (value.trim() === '') {
      problems.push(`${path}: must not be empty`);
    } else {
      for (const placeholder of shape) if (!value.includes(placeholder)) problems.push(`${path}: must contain ${placeholder}`);
    }
  } else if (!isObj(value)) {
    problems.push(`${path}: must be an object`);
  } else {
    for (const [key, child] of Object.entries(shape)) {
      const childPath = path === '' ? key : `${path}.${key}`;
      if (!Object.hasOwn(value, key)) problems.push(`${childPath}: missing`);
      else walk(child, value[key], childPath, problems);
    }
    for (const key of Object.keys(value)) {
      if (!Object.hasOwn(shape, key)) problems.push(`${path === '' ? key : `${path}.${key}`}: unknown key`);
    }
  }
}

function checkFooter(value: unknown, problems: string[]): void {
  if (!Array.isArray(value)) {
    problems.push('footer: must be an array');
    return;
  }
  value.forEach((entry: unknown, i) => {
    const path = `footer[${i}]`;
    if (!isObj(entry)) {
      problems.push(`${path}: must be an object`);
      return;
    }
    const text = entry['text'];
    if (typeof text !== 'string') problems.push(`${path}.text: must be a string`);
    else if (text.trim() === '') problems.push(`${path}.text: must not be empty`);
